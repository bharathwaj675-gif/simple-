import cv2
import numpy as np
import onnxruntime as ort

# ============================================================
# CONFIGURATION
# ============================================================

MODEL_PATH = "gender_net.onnx"

GENDER_LIST = ["Male", "Female"]

# ============================================================
# LOAD MODEL
# ============================================================

session = ort.InferenceSession(
    MODEL_PATH,
    providers=["CPUExecutionProvider"]
)

input_name = session.get_inputs()[0].name

print("Model loaded successfully")
print("Input name:", input_name)
print("Input shape:", session.get_inputs()[0].shape)

# ============================================================
# GENDER PREDICTION
# ============================================================

def predict_gender(face):

    # Convert BGR -> RGB
    image = cv2.cvtColor(face, cv2.COLOR_BGR2RGB)

    # Resize to model input
    image = cv2.resize(image, (224, 224))

    # Mean subtraction
    image = image.astype(np.float32)
    image = image - np.array([104, 117, 123], dtype=np.float32)

    # HWC -> CHW
    image = np.transpose(image, (2, 0, 1))

    # Add batch dimension
    image = np.expand_dims(image, axis=0)

    # Prediction
    output = session.run(
        None,
        {input_name: image}
    )

    probabilities = output[0][0]

    gender_index = np.argmax(probabilities)
    gender = GENDER_LIST[gender_index]

    confidence = float(probabilities[gender_index])

    return gender, confidence


# ============================================================
# WEBCAM
# ============================================================

face_cascade = cv2.CascadeClassifier(
    cv2.data.haarcascades +
    "haarcascade_frontalface_default.xml"
)

cap = cv2.VideoCapture(0)

while True:

    ret, frame = cap.read()

    if not ret:
        print("Camera error")
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    faces = face_cascade.detectMultiScale(
        gray,
        scaleFactor=1.1,
        minNeighbors=5,
        minSize=(60, 60)
    )

    for (x, y, w, h) in faces:

        face = frame[y:y+h, x:x+w]

        if face.size == 0:
            continue

        gender, confidence = predict_gender(face)

        label = f"{gender} {confidence:.1%}"

        cv2.rectangle(
            frame,
            (x, y),
            (x+w, y+h),
            (0, 255, 0),
            2
        )

        cv2.putText(
            frame,
            label,
            (x, y-10),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.7,
            (0, 255, 0),
            2
        )

    cv2.imshow("Gender Detection", frame)

    # Press Q to quit
    if cv2.waitKey(1) & 0xFF == ord("q"):
        break

cap.release()
cv2.destroyAllWindows()
